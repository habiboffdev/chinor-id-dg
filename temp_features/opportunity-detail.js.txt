import './utils.js';
import './api.js';
import './auth.js';
import './components.js';
import './navbar.js';

// assets/js/opportunity-detail.js

document.addEventListener('DOMContentLoaded', () => {
    const container = document.getElementById('opportunity-detail-container');
    const skeleton = document.getElementById('loading-skeleton');

    // State object for the sidebar actions
    const sidebarState = {
        isLoading: true,
        isLoggedIn: false,
        isStudent: false,
        canApply: false,
        hasApplied: false,
        applicationStatus: null,
        isSaved: false,
        opportunityId: null
    };

    const getOpportunityId = () => {
        const params = new URLSearchParams(window.location.search);
        sidebarState.opportunityId = params.get('id');
        return sidebarState.opportunityId;
    };

    const renderSidebarActions = () => {
        const actionsContainer = document.getElementById('sidebar-actions-container');
        if (!actionsContainer) return;

        let html = '';

        if (sidebarState.isLoading) {
            html = `<div class="btn btn-primary disabled">Loading...</div>`;
        } else if (sidebarState.hasApplied) {
            html = `
                <div class="application-status-card">
                    <div class="status-icon"><i class="fas fa-check-circle"></i></div>
                    <h3 class="status-title">Application Submitted</h3>
                    <p class="status-description">You have already applied to this opportunity.</p>
                    <div class="status-details">
                        <div class="status-item">
                            <span class="status-label">Status:</span>
                            <span class="status-badge status-${sidebarState.applicationStatus}">${formatStatusName(sidebarState.applicationStatus)}</span>
                        </div>
                    </div>
                    <a href="applications.html" class="btn btn-secondary"><i class="fas fa-eye"></i> View Application Status</a>
                </div>
            `;
        } else if (sidebarState.isLoggedIn && sidebarState.isStudent) {
            html = `
                <button id="apply-btn" class="btn btn-primary" ${!sidebarState.canApply ? 'disabled' : ''}>
                    ${sidebarState.canApply ? 'Apply Now' : 'Applications Closed'}
                </button>
                <button id="save-btn" class="btn btn-secondary" style="margin-top: 1rem;">
                    <i class="fas fa-bookmark"></i> ${sidebarState.isSaved ? 'Saved' : 'Save for Later'}
                </button>
            `;
        } else {
            html = `<a href="login.html" class="btn btn-primary">Login to Apply</a>`;
        }

        actionsContainer.innerHTML = html;

        // Add event listeners after rendering
        if (!sidebarState.hasApplied && sidebarState.isLoggedIn && sidebarState.isStudent) {
            const applyBtn = document.getElementById('apply-btn');
            if (applyBtn && sidebarState.canApply) {
                applyBtn.addEventListener('click', () => openApplicationModal(window.currentOpportunityData));
            }

            const saveBtn = document.getElementById('save-btn');
            if (saveBtn) {
                if (sidebarState.isSaved) {
                    saveBtn.disabled = true;
                }
                saveBtn.addEventListener('click', handleSaveClick);
            }
        }
    };
    
    const handleSaveClick = () => {
        const savedOpportunities = JSON.parse(localStorage.getItem('savedOpportunities') || '[]');
        if (!savedOpportunities.includes(sidebarState.opportunityId)) {
            savedOpportunities.push(sidebarState.opportunityId);
            localStorage.setItem('savedOpportunities', JSON.stringify(savedOpportunities));
            sidebarState.isSaved = true;
            Utils.showToast('Opportunity saved for later!', 'success');
        } else {
            Utils.showToast('Opportunity already saved', 'info');
        }
        renderSidebarActions(); // Re-render to update button state
    };


    const fetchOpportunityDetails = async (id) => {
        try {
            const response = await api.opportunities.getById(id);
            return response;
        } catch (error) {
            console.error('Error fetching opportunity details:', error);
            container.innerHTML = `<p class="error">Failed to load opportunity details.</p>`;
            skeleton.style.display = 'none';
            throw error; // re-throw to be caught by the caller
        }
    };

    const checkApplicationStatus = async (opportunityId) => {
        try {
            if (!auth.isLoggedIn() || auth.getCurrentUser()?.user_type !== 'student') {
                sidebarState.hasApplied = false;
                return;
            }
            
            const result = await api.applications.checkStatus(opportunityId);
            
            if (result) {
                sidebarState.hasApplied = true;
                sidebarState.applicationStatus = result.status;
            } else {
                sidebarState.hasApplied = false;
            }
        } catch (error) {
            console.error('Failed to check application status:', error);
            sidebarState.hasApplied = false;
        }
    };

    const formatDate = (dateString) => {
        if (!dateString) return 'N/A';
        const options = { year: 'numeric', month: 'long', day: 'numeric' };
        return new Date(dateString).toLocaleDateString(undefined, options);
    };

    const formatDateTime = (dateString) => {
        if (!dateString) return 'N/A';
        const date = new Date(dateString);
        const dateOptions = { year: 'numeric', month: 'long', day: 'numeric' };
        const timeOptions = { hour: '2-digit', minute: '2-digit', hour12: true };
        const formattedDate = date.toLocaleDateString(undefined, dateOptions);
        const formattedTime = date.toLocaleTimeString(undefined, timeOptions);
        return `${formattedDate} at ${formattedTime}`;
    };

    const getTimelineStatus = (date, isDeadline = false) => {
        if (!date) return 'future';
        
        const now = new Date();
        const targetDate = new Date(date);
        
        if (isDeadline) {
            if (targetDate < now) return 'overdue';
            const timeDiff = targetDate - now;
            const daysDiff = Math.ceil(timeDiff / (1000 * 60 * 60 * 24));
            if (daysDiff <= 7) return 'current';
            return 'future';
        }
        
        if (targetDate < now) return 'past';
        const timeDiff = targetDate - now;
        const daysDiff = Math.ceil(timeDiff / (1000 * 60 * 60 * 24));
        if (daysDiff <= 7) return 'current';
        return 'future';
    };

    const generateTimeline = (data) => {
        const now = new Date();
        const timeline = [];

        if (data.application_deadline) {
            timeline.push({
                date: data.application_deadline,
                title: 'Application Deadline',
                subtitle: 'Last day to submit applications',
                icon: 'fas fa-clock',
                type: 'deadline',
                status: getTimelineStatus(data.application_deadline, true)
            });
        }

        if (data.result_announcement_date) {
            timeline.push({
                date: data.result_announcement_date,
                title: 'Results Announced',
                subtitle: 'Application results will be released',
                icon: 'fas fa-bullhorn',
                type: 'announcement',
                status: getTimelineStatus(data.result_announcement_date)
            });
        }

        if (data.start_date) {
            timeline.push({
                date: data.start_date,
                title: 'Program Starts',
                subtitle: 'Opportunity/program begins',
                icon: 'fas fa-play',
                type: 'start',
                status: getTimelineStatus(data.start_date)
            });
        }

        if (data.end_date) {
            timeline.push({
                date: data.end_date,
                title: 'Program Ends',
                subtitle: 'Opportunity/program concludes',
                icon: 'fas fa-flag-checkered',
                type: 'end',
                status: getTimelineStatus(data.end_date)
            });
        }

        timeline.sort((a, b) => new Date(a.date) - new Date(b.date));
        return timeline;
    };

    const renderTimeline = (timelineData) => {
        if (timelineData.length === 0) return '';

        const timelineItems = timelineData.map((item, index) => {
            const isLeft = index % 2 === 0;
            const position = isLeft ? 'left' : 'right';
            
            return `
                <div class="timeline-item ${position} ${item.status}">
                    <div class="timeline-content ${item.status}">
                        <div class="timeline-title">
                            <i class="${item.icon} timeline-icon"></i>
                            ${item.title}
                        </div>
                        <div class="timeline-date">${formatDateTime(item.date)}</div>
                        <div class="timeline-subtitle">${item.subtitle}</div>
                    </div>
                    <div class="timeline-point"></div>
                </div>
            `;
        }).join('');

        return `
            <section class="timeline-section">
                <h2 class="section-title">
                    <i class="fas fa-calendar-alt" style="margin-right: 0.5rem;"></i>
                    Important Dates
                </h2>
                <div class="timeline">
                    ${timelineItems}
                </div>
            </section>
        `;
    };

    const renderQuestion = (q) => {
        const isRequired = q.is_required ? 'required' : '';
        const placeholder = q.placeholder ? `placeholder="${q.placeholder}"` : '';
        let inputHtml = '';

        switch (q.question_type) {
            case 'textarea':
                inputHtml = `<textarea id="question-${q.id}" class="input" data-question-id="${q.id}" ${placeholder} ${isRequired}></textarea>`;
                break;
            case 'select':
                if (q.options && q.options.length > 0) {
                    const optionsHtml = q.options.map(opt => `<option value="${opt}">${opt}</option>`).join('');
                    inputHtml = `<select id="question-${q.id}" class="input" data-question-id="${q.id}" ${isRequired}><option value="">Select...</option>${optionsHtml}</select>`;
                } else {
                    inputHtml = `<input type="text" id="question-${q.id}" class="input" data-question-id="${q.id}" ${placeholder} ${isRequired}>`;
                }
                break;
            default:
                inputHtml = `<input type="${q.question_type || 'text'}" id="question-${q.id}" class="input" data-question-id="${q.id}" ${placeholder} ${isRequired}>`;
        }

        return `
            <div class="form-group">
                <label for="question-${q.id}">${q.question}${q.is_required ? '*' : ''}</label>
                ${inputHtml}
            </div>
        `;
    };

    const renderOpportunity = async (data) => {
        skeleton.style.display = 'none';
        window.currentOpportunityData = data; // Store for modal access

        const coverImageContainer = document.getElementById('opportunity-cover-image');
        if (coverImageContainer && data.cover_image) {
            coverImageContainer.style.backgroundImage = `url(${data.cover_image})`;
        }
        
        await auth.waitForUser();
        
        const currentUser = auth.getCurrentUser();
        sidebarState.isLoggedIn = auth.isLoggedIn();
        sidebarState.isStudent = currentUser?.user_type === 'student';
        sidebarState.canApply = data.can_apply;

        await checkApplicationStatus(data.id);

        const savedOpportunities = JSON.parse(localStorage.getItem('savedOpportunities') || '[]');
        sidebarState.isSaved = savedOpportunities.includes(data.id);
        
        sidebarState.isLoading = false;

        const html = `
            <div class="opportunity-main">
                <section class="section">
                    <div class="opportunity-header">
                        <h1 class="opportunity-title">${data.title}</h1>
                        <p class="organization-name">at ${data.organization.name}</p>
                        <div class="tags">
                            <span class="tag tag-type">${data.opportunity_type}</span>
                            <span class="tag tag-status">${data.status}</span>
                            ${data.is_remote ? '<span class="tag tag-remote">Remote</span>' : ''}
                        </div>
                    </div>
                </section>

                <section class="section">
                    <h2 class="section-title">Description</h2>
                    <div class="description-content">
                        ${data.description.replace(/\n/g, '<br>')}
                    </div>
                </section>

                ${renderTimeline(generateTimeline(data))}

                <section class="section">
                    <h2 class="section-title">Details</h2>
                    <div class="details-grid">
                        <div class="detail-item">
                            <div class="detail-item-label">Location</div>
                            <div class="detail-item-value">${data.location}</div>
                        </div>
                        <div class="detail-item">
                            <div class="detail-item-label">Compensation</div>
                            <div class="detail-item-value">${data.compensation || 'N/A'}</div>
                        </div>
                    </div>
                </section>

                <section class="section">
                    <h2 class="section-title">Requirements</h2>
                    <div class="details-grid">
                        ${data.min_gpa ? `<div class="detail-item"><div class="detail-item-label">Minimum GPA</div><div class="detail-item-value">${data.min_gpa}</div></div>` : ''}
                        ${data.required_major ? `<div class="detail-item"><div class="detail-item-label">Required Major</div><div class="detail-item-value">${data.required_major}</div></div>` : ''}
                        ${(data.graduation_year_min || data.graduation_year_max) ? `<div class="detail-item"><div class="detail-item-label">Graduation Year</div><div class="detail-item-value">${data.graduation_year_min || ''} - ${data.graduation_year_max || ''}</div></div>` : ''}
                        ${(data.age_min || data.age_max) ? `<div class="detail-item"><div class="detail-item-label">Age Requirement</div><div class="detail-item-value">${data.age_min ? `${data.age_min}+` : ''}${data.age_min && data.age_max ? ' - ' : ''}${data.age_max ? `${data.age_max} years` : ''}</div></div>` : ''}
                    </div>
                </section>

                ${data.required_skills.length > 0 ? `<section class="section"><h2 class="section-title">Required Skills</h2><div class="skills-list">${data.required_skills.map(skill => `<span class="skill-tag">${skill.name}</span>`).join('')}</div></section>` : ''}

                ${data.profile_requirements && data.profile_requirements.length > 0 ? `<section class="section"><h2 class="section-title"><i class="fas fa-user-check" style="margin-right: 0.5rem; color: var(--accent-1);"></i>Required Profile Sections</h2><div class="profile-requirements">${data.profile_requirements.map(req => {
                    const levelClass = req.requirement_level === 'required' ? 'required' : req.requirement_level === 'recommended' ? 'recommended' : 'optional';
                    const levelIcon = req.requirement_level === 'required' ? 'fas fa-exclamation-circle' : req.requirement_level === 'recommended' ? 'fas fa-star' : 'fas fa-circle';
                    return `<div class="profile-requirement ${levelClass}"><div class="requirement-header"><i class="${levelIcon}"></i><span class="requirement-section">${req.section_display || req.section}</span><span class="requirement-level">${req.requirement_level}</span></div>${req.custom_message ? `<div class="requirement-message">${req.custom_message}</div>` : ''}${req.minimum_items > 1 ? `<div class="requirement-min">Minimum ${req.minimum_items} items required</div>` : ''}</div>`;
                }).join('')}</div></section>` : ''}
                
                ${data.benefits ? `<section class="section"><h2 class="section-title"><i class="fas fa-gift" style="margin-right: 0.5rem; color: var(--accent-1);"></i>Benefits & Perks</h2><div class="description-content benefits-content">${data.benefits.replace(/\n/g, '<br>')}</div></section>` : ''}

                ${data.application_instructions ? `<section class="section application-guidelines"><h2 class="section-title"><i class="fas fa-clipboard-list" style="margin-right: 0.5rem; color: var(--accent-2);"></i>Application Guidelines</h2><div class="description-content guidelines-content">${data.application_instructions.replace(/\n/g, '<br>')}</div></section>` : ''}
            </div>

            <aside class="opportunity-sidebar">
                <div class="sidebar-card" id="sidebar-actions-container">
                    <!-- Sidebar actions will be rendered here by renderSidebarActions() -->
                </div>
                <div class="sidebar-card">
                    <h3 class="sidebar-title">Key Information</h3>
                    <ul class="info-list">
                        <li><i class="fas fa-calendar-times"></i><div class="info-item-content"><strong>Application Deadline:</strong><div class="deadline-datetime">${formatDateTime(data.application_deadline)}</div></div></li>
                        <li><i class="fas fa-users"></i> <strong>Applications:</strong> ${data.application_count} received</li>
                        ${data.max_applications ? `<li><i class="fas fa-bullseye"></i> <strong>Limit:</strong> ${data.max_applications} applications</li>` : ''}
                        ${data.result_announcement_date ? `<li><i class="fas fa-bullhorn"></i><div class="info-item-content"><strong>Results Announced:</strong><div class="announcement-datetime">${formatDateTime(data.result_announcement_date)}</div></div></li>` : ''}
                    </ul>
                </div>
                <div class="sidebar-card organization-card">
                    <h3 class="sidebar-title"><i class="fas fa-building" style="margin-right: 0.5rem; color: var(--accent-1);"></i>About ${data.organization.name}</h3>
                    <div class="organization-content">
                        ${data.organization.bio ? `<p class="organization-bio">${data.organization.bio}</p>` : `<p class="organization-bio">This organization provides exciting opportunities for students to grow and develop their skills.</p>`}
                        ${data.organization.industry ? `<div class="org-detail"><i class="fas fa-industry"></i><span><strong>Industry:</strong> ${data.organization.industry}</span></div>` : ''}
                        ${data.organization.location ? `<div class="org-detail"><i class="fas fa-map-marker-alt"></i><span><strong>Location:</strong> ${data.organization.location}</span></div>` : ''}
                        ${data.organization.founded_year ? `<div class="org-detail"><i class="fas fa-calendar-alt"></i><span><strong>Founded:</strong> ${data.organization.founded_year}</span></div>` : ''}
                        <div class="organization-actions">
                            ${data.organization.website ? `<a href="${data.organization.website}" target="_blank" class="btn btn-secondary"><i class="fas fa-external-link-alt"></i> Visit Website</a>` : ''}
                            ${data.organization.linkedin ? `<a href="${data.organization.linkedin}" target="_blank" class="btn btn-ghost"><i class="fab fa-linkedin"></i> LinkedIn</a>` : ''}
                        </div>
                    </div>
                </div>
            </aside>
        `;

        container.innerHTML = html;
        renderSidebarActions(); // Initial render of the sidebar actions
    };

    const openApplicationModal = (opportunityData) => {
        const modal = document.getElementById('applicationModal');
        const title = document.getElementById('modal-title');
        const questionsContainer = document.getElementById('additional-questions-container');

        if (modal && title && questionsContainer) {
            title.textContent = `${opportunityData.title} at ${opportunityData.organization.name}`;
            
            questionsContainer.innerHTML = '';
            if (opportunityData.additional_questions && opportunityData.additional_questions.length > 0) {
                questionsContainer.innerHTML = opportunityData.additional_questions.map(renderQuestion).join('');
            }

            modal.style.display = 'flex';
            setupFileUpload();
        }

        document.getElementById('closeApplicationModal').addEventListener('click', closeApplicationModal);
        document.getElementById('cancelApplication').addEventListener('click', closeApplicationModal);
    };

    const setupFileUpload = () => {
        const uploadArea = document.getElementById('documentUploadArea');
        const fileInput = document.getElementById('documentUpload');
        const filesList = document.getElementById('uploaded-files-list');

        if (!uploadArea || !fileInput) return;

        uploadArea.addEventListener('click', () => fileInput.click());
        uploadArea.addEventListener('dragover', (e) => { e.preventDefault(); uploadArea.style.borderColor = 'var(--accent-2)'; });
        uploadArea.addEventListener('dragleave', () => { uploadArea.style.borderColor = 'var(--slate-400)'; });
        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            uploadArea.style.borderColor = 'var(--slate-400)';
            fileInput.files = e.dataTransfer.files;
            updateFilesList();
        });
        fileInput.addEventListener('change', updateFilesList);

        function updateFilesList() {
            if (!filesList) return;
            filesList.innerHTML = '';
            Array.from(fileInput.files).forEach((file, index) => {
                const fileItem = document.createElement('div');
                fileItem.className = 'file-item';
                fileItem.innerHTML = `<div class="file-info"><i class="fas fa-file"></i><span>${file.name}</span><span class="file-size">(${Utils.formatFileSize(file.size)})</span></div><button type="button" class="file-remove" data-index="${index}"><i class="fas fa-times"></i></button>`;
                filesList.appendChild(fileItem);
            });
            filesList.querySelectorAll('.file-remove').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    const index = parseInt(e.currentTarget.dataset.index);
                    const dt = new DataTransfer();
                    Array.from(fileInput.files).forEach((file, i) => { if (i !== index) dt.items.add(file); });
                    fileInput.files = dt.files;
                    updateFilesList();
                });
            });
        }
    };

    const closeApplicationModal = () => {
        const modal = document.getElementById('applicationModal');
        if (modal) modal.style.display = 'none';
    };

    const handleApplicationSubmit = async (event) => {
        event.preventDefault();
        const submitBtn = document.getElementById('submitApplication');
        const submitText = document.getElementById('submitButtonText');
        const spinner = document.getElementById('submitSpinner');

        submitBtn.disabled = true;
        submitText.textContent = 'Submitting...';
        spinner.classList.remove('hidden');

        try {
            const opportunityId = getOpportunityId();
            const notes = document.getElementById('applicationNotes')?.value || '';
            const files = document.getElementById('documentUpload').files;

            const additionalAnswers = [];
            document.querySelectorAll('#additional-questions-container [data-question-id]').forEach(input => {
                if (input.value.trim()) {
                    additionalAnswers.push({
                        question: parseInt(input.dataset.questionId),
                        answer_text: input.value
                    });
                }
            });

            const applicationData = {
                opportunity: parseInt(opportunityId),
                notes: notes,
                answers: additionalAnswers
            };

            const response = await api.applications.submit(applicationData);

            Utils.showToast('Application submitted successfully!', 'success');
            closeApplicationModal();
            
            // Update state and re-render sidebar
            sidebarState.hasApplied = true;
            sidebarState.applicationStatus = 'pending'; // Assume pending status on successful submission
            renderSidebarActions();

        } catch (error) {
            console.error('Application submission failed:', error);
            // Error is already shown by the global api handler
        } finally {
            submitBtn.disabled = false;
            submitText.textContent = 'Submit Application';
            spinner.classList.add('hidden');
        }
    };

    const init = async () => {
        const opportunityId = getOpportunityId();
        if (!opportunityId) {
            container.innerHTML = `<div class="error"><i class="fas fa-exclamation-triangle"></i><h3>No Opportunity Found</h3><p>No opportunity ID was provided in the URL.</p><a href="opportunities.html" class="btn btn-secondary">Browse Opportunities</a></div>`;
            skeleton.style.display = 'none';
            return;
        }
        
        renderSidebarActions(); // Initial render for loading state

        try {
            const opportunityData = await fetchOpportunityDetails(opportunityId);
            if (opportunityData) {
                renderOpportunity(opportunityData);
                document.getElementById('applicationForm').addEventListener('submit', handleApplicationSubmit);
            }
        } catch (error) {
            console.error('Failed to initialize opportunity detail page:', error);
            container.innerHTML = `<div class="error"><i class="fas fa-exclamation-triangle"></i><h3>Failed to Load Opportunity</h3><p>There was an error loading this opportunity. Please try again.</p><button onclick="window.location.reload()" class="btn btn-secondary">Retry</button></div>`;
            skeleton.style.display = 'none';
        }
    };

    init();
});
